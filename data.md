    const searchInput = document.getElementById('table-search');
    const modelFilter = document.getElementById('model-filter');
    const body = document.getElementById('table-body');
    const count = document.getElementById('result-count');
    const noResults = document.getElementById('no-results');
    let sortColumn = -1;
    let sortDirection = 1;

    function textCell(value) {
      const td = document.createElement('td');
      if (value === null || value === undefined || String(value).trim() === '') {
        td.textContent = '\u2014';
        td.className = 'empty';
      } else {
        td.textContent = String(value);
      }
      return td;
    }

    function getFilteredRows() {
      const query = searchInput.value.trim().toLocaleLowerCase();
      const model = modelFilter.value;
      let result = rows.filter(row => {
        const text = row.join(' ').toLocaleLowerCase();
        return (!query || text.includes(query)) && (!model || row[0] === model);
      });
      if (sortColumn >= 0) {
        result = result.slice().sort((a, b) => {
          const av = String(a[sortColumn] ?? '').toLocaleLowerCase();
          const bv = String(b[sortColumn] ?? '').toLocaleLowerCase();
          return av.localeCompare(bv, undefined, { numeric: true, sensitivity: 'base' }) * sortDirection;
        });
      }
      return result;
    }

    function render() {
      const visible = getFilteredRows();
      body.replaceChildren();
      const fragment = document.createDocumentFragment();
      for (const row of visible) {
        const tr = document.createElement('tr');
        row.forEach(value => tr.appendChild(textCell(value)));
        fragment.appendChild(tr);
      }
      body.appendChild(fragment);
      count.textContent = `${visible.length} of ${rows.length} entries shown`;
      noResults.hidden = visible.length !== 0;
    }

    document.querySelectorAll('#table-head button').forEach((button, index) => {
      button.addEventListener('click', () => {
        if (sortColumn === index) sortDirection *= -1;
        else { sortColumn = index; sortDirection = 1; }
        headers.forEach((_, i) => {
          const b = document.querySelector(`#table-head th:nth-child(${i + 1}) button`);
          if (b) {
            b.textContent = headers[i] + (sortColumn === i ? (sortDirection === 1 ? ' â–²' : ' â–¼') : '');
            b.setAttribute('aria-label', `${headers[i]}${sortColumn === i ? (sortDirection === 1 ? ', ascending' : ', descending') : ', sort'}`);
          }
        });
        render();
      });
    });

    [...new Set(rows.map(row => row[0]))].sort((a, b) => a.localeCompare(b)).forEach(model => {
      const option = document.createElement('option');
      option.value = model;
      option.textContent = model;
      modelFilter.appendChild(option);
    });
    searchInput.addEventListener('input', render);
    modelFilter.addEventListener('change', render);
    render();
